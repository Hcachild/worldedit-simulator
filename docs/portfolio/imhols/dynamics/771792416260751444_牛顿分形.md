## 动态内容

牛顿分形
z->z-(z^3-1)/(3*z^2)

//generate 251 
x*=2;z*=2;
for(n=0,10){
k=(x^2-z^2)^2+4*(x*z)^2;
nx=(x*2+(x^2-z^2)/k)/3;nz=(z*2-2*x*z/k)/3;
x=nx;z=nz
};
r1=(x-1)^2+z^2;
r2=(x+1/2)^2+(z-sqrt(3)/2)^2;
r3=(x+1/2)^2+(z+sqrt(3)/2)^2;
data=r1<r2&&r1<r3?1:(r2<r3?2:3)
