## 动态内容

希尔伯特楼梯

size:100,256,100

//g 251 
n=4;h=0;f=0;
for(i=1,n){
Z=0;X=0;
a=(z-Z>0);b=(x-X>0);
f=a*b?2:!a*b?1:!a*!b?0:3;
h+=4^(n-i)*f;
Z-=(-1)^a/2^i;X-=(-1)^b/2^i;
z-=Z;x-=X;
if(f==0){rotate(x,z,pi/2)x=-x;}
if(f==3){rotate(x,z,-pi/2)x=-x;}
}
y+1<h/2^(2*n-1)
